\newcommand{\ThmStmtNGARROWDPI}{%
Let $X$ and $Y$ be finite sets, let $\mu\in\Delta(X)$, let $P$ be a Markov kernel on $X$, let $f:X\to Y$ be a lens, and let $T\ge 0$. Then
\[
  \Arr_T(\mu,P;f)\;\le\;\Arr_T(\mu,P),
\]
that is,
\[
  \DKL{f^{(T)}_{\#}\mathbf{P}^{\mu,P}_T}{\rev_{\#}f^{(T)}_{\#}\mathbf{P}^{\mu,P}_T}\;\le\;\DKL{\mathbf{P}^{\mu,P}_T}{\rev_{\#}\mathbf{P}^{\mu,P}_T},
\]
as an inequality in $[0,\infty]$.
}

\newcommand{\ThmStmtNGPROTOCOLTRAP}{%
Let $P$ be a Markov kernel on a finite set $X$, and let $\pi\in\Delta(X)$ satisfy detailed balance, $\pi(x)P(x,x')=\pi(x')P(x',x)$ for all $x,x'\in X$. Then for every horizon $T\ge 0$ the path law is reversal invariant,
\[
  \rev_{\#}\mathbf{P}^{\pi,P}_T=\mathbf{P}^{\pi,P}_T,
\]
so $\Arr_T(\pi,P)=0$. Consequently $\Arr_T(\pi,P;f)=0$ for every lens $f:X\to Y$ into a finite set $Y$.
}

\newcommand{\ThmStmtNGFORCEFOREST}{%
Let $G=(V,E)$ be a forest, not necessarily finite, and let $a$ be an antisymmetric edge field on $G$. Then $a$ is exact: there is $\phi:V\to\mathbb{R}$ with
\[
  a(u,v)=\phi(v)-\phi(u)\qquad\text{for every edge }\{u,v\}\in E.
\]
}

\newcommand{\ThmStmtNGFORCENULL}{%
Let $G=(V,E)$ be a graph, let $\phi:V\to\mathbb{R}$, and let $a$ be an edge field with $a(u,v)=\phi(v)-\phi(u)$ for every edge $\{u,v\}\in E$. Then for every closed walk $v_0,v_1,\dots,v_k=v_0$ in $G$,
\[
  \sum_{i=0}^{k-1}a(v_i,v_{i+1})=0.
\]
}

\newcommand{\ThmStmtNGMACROCLOSUREDEFICIT}{%
Let $P$ be a Markov kernel on a finite set $X$, let $f:X\to Y$ be a lens into a finite set, let $\tau\ge 0$, and let $\mu\in\Delta(X)$ be the current law. Define $K^\star$ by
\[
  K^\star(y,\cdot)=\Law\bigl(Y_\tau\bigm| Y_0=y\bigr)=\frac{1}{\mu(f^{-1}(y))}\sum_{x\in f^{-1}(y)}\mu(x)\,p_x
  \qquad\text{if }\mu(f^{-1}(y))>0,
\]
and let $K^\star(y,\cdot)$ be an arbitrary law if $\mu(f^{-1}(y))=0$. Then:
\begin{enumerate}[label=(\roman*)]
  \item $\mathcal{L}_\tau(K^\star)=\mathrm{CD}_\tau(\mu;P,f)$;
  \item $\mathrm{CD}_\tau(\mu;P,f)\le\mathcal{L}_\tau(K)$ for every macro kernel $K$, including kernels with $\mathcal{L}_\tau(K)=\infty$; hence $\mathrm{CD}_\tau(\mu;P,f)=\min_K\mathcal{L}_\tau(K)$;
  \item $\mathrm{CD}_\tau(\mu;P,f)=0$ if and only if $p_x=p_{x'}$ whenever $f(x)=f(x')$, $\mu(x)>0$ and $\mu(x')>0$. In particular, if two microstates of positive mass in the same fiber have different packaged future laws, then $\mathrm{CD}_\tau(\mu;P,f)>0$.
\end{enumerate}
}

\newcommand{\ThmStmtNGOBJECTCONTRACTIVE}{%
Let $P$ be a Markov kernel on a nonempty finite set $S$ with Dobrushin coefficient $\lambda(P)<1$, and let $\varepsilon\ge 0$. If $\nu,\nu'\in\Delta(S)$ are $\varepsilon$-stable, that is, $\dTV(\nu,\nu P)\le\varepsilon$ and $\dTV(\nu',\nu'P)\le\varepsilon$, then
\[
  \dTV(\nu,\nu')\;\le\;\frac{2\varepsilon}{1-\lambda(P)}.
\]
In particular, $P$ has at most one stationary law.
}

\newcommand{\ThmStmtNGLADDERIDEM}{%
Let $S$ be any set and let $e:S\to S$ satisfy $e\circ e=e$. Then $e^{\circ n}=e$ for every integer $n\ge 1$.
}

\newcommand{\ThmStmtNGLADDERBOUNDEDINTERFACE}{%
Let $f:X\to Y$ be a map between arbitrary sets whose image $\im(f)$ is finite. Then
\[
  |\Def(f)|=2^{|\im(f)|}.
\]
Consequently there is no infinite sequence $h_1,h_2,\dots$ of pairwise distinct $f$-definable predicates; in particular there is no infinite strictly increasing chain of them.
}
